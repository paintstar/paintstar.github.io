## <a id="D__2"></a>D ：

画个图，纯数学题，想要确定方向的时候应该联想到使用向量，精度问题注意用两者相减大于eps来确定。已知两点坐标求向量为 `x = x2-x1 , y = y2-y1`; 两个向量相乘为 `x1* x2 + y1* y2`;把范围缩小到0-180度cos值单调递减。

```cpp
#include<bits/stdc++.h>
using namespace std;

#define ll long long
double cal(double x1, double y1, double x2, double y2)
{
    double x = (x1 - x2) * (x1 - x2);
    double y = (y1 - y2) * (y1 - y2);
    double res = sqrt(x + y);
    return res;
}
struct vec{
    double x, y;
};

int main()
{
    ios::sync_with_stdio(false); cin.tie(0); cout.tie(0);
    int t;
    cin >> t;
    while (t--)
    {
        double x1, y1, x2, y2, x3, y3;
        cin >> x1 >> y1 >> x2 >> y2 >> x3 >> y3;
        double d12 = cal(x1, y1, x2, y2);
        double d23 = cal(x2, y2, x3, y3);
        double cosa = 2 / d12;
        double sinb = 2 / d23;
        double cosb = 1 - sinb * sinb;
        double sina = 1 - cosa * cosa;
        double cosa_b = cosa * cosb - sina * sinb; 
        vec a = { x2 - x1,y2 - y1 };
        vec b = { x3 - x2,y3 - y2 };
        double cosc = (a.x * b.x + a.y * b.y) / (d12 * d23);
        if (cosc - cosa_b > 1e-8) cout << "yes\n";
        else cout << "no\n";
    }
    cout << endl;
}
```
