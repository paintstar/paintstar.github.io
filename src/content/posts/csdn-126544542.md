[题目传送门](https://codeforces.com/problemset/problem/1077/D)

> 二分找到最多能分多少组，再根据能分多少组找到每组分别分谁

## <a id="AC_3"></a>AC代码：

```cpp
#include<bits/stdc++.h>
using namespace std;
#define endl "\n";
#define ll long long
#define inf 0x3f3f3f3f
#define pii pair<int,int>
const int N = 2e5+5;
map<int,int>mp;
int a[N],b[N];
int n, k;
vector<int>ans;
set<int>st;
bool check(int x,int n)
{
	int ff = 1;
	for (int i = n; i >= 1; i--)
	{
		if (b[i] < x) ff = i + 1;
	}
	ll sum = 0;
	for (int i = ff; i <= n; i++)
	{
		sum += (b[i] - b[i] % x);
	}
	if (sum >= (ll)k * x) return 1;
	else return 0;
}
int main()
{
	ios::sync_with_stdio(false); cin.tie(0); cout.tie(0);
	cin >> n >> k;
	int cnt=0;
	for (int i = 1; i <= n; i++) cin >> a[i],mp[a[i]]++,st.insert(a[i]);
	for (auto& i : st)
	{
		b[++cnt] = mp[i];
	}
	sort(b + 1, b + 1 + cnt);
	int l = 1, r = b[cnt];
	while (l < r)
	{
		int mid = (l + r+1) >> 1;
		if (check(mid,cnt)) l=mid;
		else r=mid-1;
	}
	for (int i = 1; i <= n&&k; i++)
	{
		if(mp[a[i]]>=l)
		{
			while (mp[a[i]] >= l)
			{
				ans.push_back(a[i]);
				mp[a[i]] -= l;
				k--;
				if (!k) break;
			}
		}
	}
	for (auto& i : ans)
		cout << i << ' ';
	return 0;
}
```
